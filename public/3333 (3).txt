
---
import { getCollection } from "astro:content";
import BaseLayout from "../layouts/BaseLayout.astro";

const base = import.meta.env.BASE_URL;

const projects = await getCollection(
  "projects",
  ({ data }) => !data.draft
);

const sortedProjects = projects.sort((a, b) => {
  const order = {
    Active: 0,
    Planned: 1,
    "On Hold": 2,
    Completed: 3,
  };

  return order[a.data.status] - order[b.data.status];
});
---

<BaseLayout
  title="Research Projects | Trace & Triage"
  description="Explore technical laboratories, digital forensic research environments and ongoing Trace & Triage projects."
>

  <section class="projects-header">
    <p class="case-id">RESEARCH ENVIRONMENTS</p>

    <h1>Projects</h1>

    <p class="hero-subtitle">
      Technical laboratories, experiments and longer-term
      digital forensic research.
    </p>
  </section>

  <div class="project-grid">

    {sortedProjects.map((project) => (
      <a
        class="project-card"
        href={`${base}projects/${project.id}/`}
      >
        <div class="project-top">
          <span class="case-id">
            {project.data.projectId}
          </span>

          <span
            class:list={[
              "project-status",
              project.data.status.toLowerCase().replace(" ", "-")
            ]}
          >
            {project.data.status}
          </span>
        </div>

        <h2>{project.data.title}</h2>

        <p>{project.data.description}</p>

        <div class="project-technologies">
          {project.data.technologies.slice(0, 4).map((tech) => (
            <span>{tech}</span>
          ))}
        </div>

        <span class="project-link">
          View Project →
        </span>
      </a>
    ))}

  </div>

  {sortedProjects.length === 0 && (
    <p class="muted">
      Research projects are being prepared for publication.
    </p>
  )}

</BaseLayout>
